$ErrorActionPreference = 'Stop'

$connStr = 'Server=WMSDB;Database=master;User Id=sa;Password=<ВАШ_ПАРОЛЬ_sa>;'

$listSql = @'
CREATE TABLE #d (subdirectory NVARCHAR(512), depth INT, isfile BIT);
INSERT #d EXEC master.dbo.xp_dirtree N'\\wmsbackup\LV_MERIDIAN\', 0, 1;
SELECT subdirectory FROM #d
 WHERE isfile = 1 AND subdirectory LIKE N'%.bak'
 ORDER BY subdirectory;
'@

function Get-BackupList {
    $cn = New-Object System.Data.SqlClient.SqlConnection $connStr
    $cn.Open()
    try {
        $cmd = $cn.CreateCommand()
        $cmd.CommandText = $listSql
        $cmd.CommandTimeout = 30
        $rd = $cmd.ExecuteReader()
        $names = New-Object System.Collections.Generic.List[string]
        while ($rd.Read()) { $names.Add($rd.GetString(0)) }
        $rd.Close()
        return $names
    } finally { $cn.Close() }
}

$listener = New-Object System.Net.HttpListener
$listener.Prefixes.Add('http://+:8090/')
$listener.Start()
Write-Host 'Слушаю http://+:8090/ ...'

while ($listener.IsListening) {
    $ctx = $listener.GetContext()
    try {
        $files = Get-BackupList
        $items = ($files | ForEach-Object { "<li>$_</li>" }) -join ''
        $html = "<h1>Резервные копии LV_MERIDIAN</h1><ul>$items</ul>"
        $ctx.Response.StatusCode = 200
    } catch {
        $html = "<h1>Ошибка получения списка</h1><pre>$($_.Exception.Message)</pre>"
        $ctx.Response.StatusCode = 500
    }
    $buf = [Text.Encoding]::UTF8.GetBytes($html)
    $ctx.Response.ContentType = 'text/html; charset=utf-8'
    $ctx.Response.OutputStream.Write($buf, 0, $buf.Length)
    $ctx.Response.Close()
}
